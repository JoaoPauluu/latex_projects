\NewDocumentCommand{\asyimage}{O{width=0.8\linewidth} m m o}{%
  \begin{figure}[H]%
    \centering
    \includegraphics[#1]{#2}%
    \caption{#3}%
    \IfValueT{#4}{\label{#4}}%
  \end{figure}%
}

\newenvironment{citacao}{%
  \par\smallskip
  \begingroup
  \leftskip=4cm
  \small
  \ifdefined\singlespacing\singlespacing\fi
  \noindent\ignorespaces
}{%
  \par\endgroup
  \smallskip
}
